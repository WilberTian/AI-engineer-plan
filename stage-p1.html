P1-2 类型与 Pydantic：把 TS 的心智平移过来 40min
为什么：AI 应用的输入输出全是「模型返回的、不可信的 JSON」。TS 里你用 Zod 挡在边界上，Python 里就是 Pydantic。这个映射你必须闭眼能写，因为后面每一周都在用。
怎么做：先看映射表，再手写一个
TSPython关键差异
interface X {a: string}class X(BaseModel): a: strTS 类型在运行时消失；Pydantic 在运行时校验
z.object({a: z.string()})BaseModel 子类Pydantic 会尝试类型转换（"1" → 1），Zod 默认不转。想要严格要加 strict=True
a?: stringa: str | None = NonePython 必须给默认值；Optional 只是类型标注，不自动给 None
"a" | "b"Literal["a", "b"]约束模型只能返回这几个值，比事后判断好用
z.enum([...])enum.StrEnum 或 LiteralAI 场景优先 Literal，序列化更省事
z.array(X)list[X]3.9+ 可直接用内置泛型
z.string().min(1)Field(min_length=1)校验规则写在 Field 里
z.infer<typeof S>就是那个类本身不用单独推类型

from typing import Literal

from pydantic import BaseModel, Field, field_validator


class Citation(BaseModel):
    doc_id: str = Field(description="来源文档 ID，必须来自检索结果，不允许编造")
    quote: str = Field(min_length=1, max_length=300, description="原文片段")


class Answer(BaseModel):
    answer: str = Field(description="给用户的回答；没有依据时必须为空字符串")
    confidence: Literal["high", "medium", "low"]
    citations: list[Citation] = Field(default_factory=list)
    refused: bool = Field(default=False, description="是否因缺少依据而拒答")

    @field_validator("citations")
    @classmethod
    def need_citation_unless_refused(cls, v: list[Citation], info):
        # 关键业务规则：要么有引用，要么明确拒答。两者都不满足就是坏数据。
        if not v and not info.data.get("refused", False):
            raise ValueError("非拒答回答必须至少有一条引用")
        return v

P1-3 asyncio：coroutine 不是 Promise（本周最容易错的一天） 40min
为什么：AI 应用 90% 的时间在等网络。不会并发就等于把延迟叠加。而 Python 的异步模型和 JS 有两处根本差异，不知道会写出「看起来能跑但实际串行」或者「整个服务被一个阻塞调用卡死」的代码。
两处根本差异
行为JS / TSPython
调用 async 函数时 const p = fetch() → 立刻开始执行，返回 Promise coro = fetch() → 什么都没发生，返回 coroutine 对象。必须 await 或在 gather 里才会跑
顶层怎么启动 模块顶层 await（ESM） 必须 asyncio.run(main())，由它创建并关闭事件循环
并发 Promise.all([a, b]) await asyncio.gather(a(), b())；注意是调用后的 coroutine，不是函数本身
阻塞调用 几乎不存在这个问题（Node 全异步） 致命：在 async 函数里调 requests.get() 会卡住整个事件循环，所有并发请求一起等
超时 AbortController asyncio.timeout(30) 上下文管理器
动手：写一个「并发调 N 个模型」的对比实验
"""scripts/async_demo.py — 用耗时差异证明你理解了 coroutine"""
from __future__ import annotations

import asyncio
import time

import httpx


async def fake_llm(client: httpx.AsyncClient, idx: int) -> int:
    # 模拟 800ms 的模型延迟
    await asyncio.sleep(0.8)
    return idx


async def serial(n: int) -> float:
    t0 = time.perf_counter()
    async with httpx.AsyncClient() as client:
        for i in range(n):
            await fake_llm(client, i)
    return time.perf_counter() - t0


async def concurrent(n: int) -> float:
    t0 = time.perf_counter()
    async with httpx.AsyncClient() as client:
        # 注意：这里是 fake_llm(client, i)，先创建 coroutine 再交给 gather
        await asyncio.gather(*(fake_llm(client, i) for i in range(n)))
    return time.perf_counter() - t0


async def with_timeout() -> str:
    try:
        async with asyncio.timeout(0.3):
            await asyncio.sleep(1)
    except TimeoutError:
        return "超时被拦住"
    return "不应该到这里"


async def main() -> None:
    n = 5
    s = await serial(n)
    c = await concurrent(n)
    print(f"串行 {n} 次: {s:.2f}s")
    print(f"并发 {n} 次: {c:.2f}s  (理论 ≈0.8s)")
    print(f"加速比: {s / c:.1f}x")
    print(await with_timeout())


if __name__ == "__main__":
    asyncio.run(main())

跑它，期望 串行 5 次: 4.00s / 并发 5 次: 0.80s / 加速比 5x。如果你的并发也是 4 秒，说明你写成了 gather(fake_llm, ...)（传了函数不是 coroutine）——这是最常见的错。

"""tools/toc.py — 用法：uv run python -m tools.toc data/ --depth 2 --json"""
from __future__ import annotations

import argparse
import json
import re
from dataclasses import dataclass, field
from pathlib import Path


@dataclass
class Node:
    level: int
    title: str
    children: list["Node"] = field(default_factory=list)


HEADING = re.compile(r"^(#{1,6})\s+(.+)$")


def parse_file(path: Path) -> list[Node]:
    roots: list[Node] = []
    stack: list[Node] = []
    for line in path.read_text(encoding="utf-8").splitlines():
        m = HEADING.match(line)
        if not m:
            continue
        node = Node(level=len(m.group(1)), title=m.group(2).strip())
        while stack and stack[-1].level >= node.level:
            stack.pop()
        if stack:
            stack[-1].children.append(node)
        else:
            roots.append(node)
        stack.append(node)
    return roots


def trim(nodes: list[Node], depth: int, cur: int = 1) -> list[Node]:
    if cur > depth:
        return []
    for n in nodes:
        n.children = trim(n.children, depth, cur + 1)
    return nodes


def render(nodes: list[Node], indent: int = 0) -> str:
    out = []
    for n in nodes:
        out.append("  " * indent + f"- {n.title}")
        out.append(render(n.children, indent + 1))
    return "\n".join(x for x in out if x)


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("path", type=Path)
    ap.add_argument("--depth", type=int, default=3)
    ap.add_argument("--json", action="store_true")
    args = ap.parse_args()

    files = sorted(args.path.rglob("*.md")) if args.path.is_dir() else [args.path]
    result = {f.name: trim(parse_file(f), args.depth) for f in files}

    if args.json:
        print(json.dumps(
            {k: [n.__dict__ for n in v] for k, v in result.items()},
            ensure_ascii=False, indent=2))
    else:
        for name, nodes in result.items():
            print(f"\n## {name}")
            print(render(nodes))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

为什么：四个方法是所有 LLM 应用的原子操作。把它们封装在一个类里，后面所有代码只调这四样——P4 做模型路由时才只需要改一个文件。
"""src/agentlab/llm.py — 唯一的模型调用出口"""
from __future__ import annotations

import json
from collections.abc import Iterator
from typing import Any, TypeVar

from openai import OpenAI
from pydantic import BaseModel, ValidationError

from agentlab.config import MAX_OUTPUT_TOKENS, TEMPERATURE, ProviderConfig

T = TypeVar("T", bound=BaseModel)


class LLM:
    def __init__(self, provider: ProviderConfig) -> None:
        self.provider = provider
        self.client = OpenAI(
            api_key=provider.api_key,
            base_url=provider.base_url,
            timeout=60,
        )

    # ---------- 1. 普通对话 ----------
    def chat(self, messages: list[dict[str, Any]], **kw: Any) -> str:
        resp = self.client.chat.completions.create(
            model=self.provider.model,
            messages=messages,  # type: ignore[arg-type]
            temperature=kw.get("temperature", TEMPERATURE),
            max_tokens=kw.get("max_tokens", MAX_OUTPUT_TOKENS),
        )
        return resp.choices[0].message.content or ""

    # ---------- 2. 流式 ----------
    def stream(self, messages: list[dict[str, Any]]) -> Iterator[str]:
        stream = self.client.chat.completions.create(
            model=self.provider.model,
            messages=messages,  # type: ignore[arg-type]
            temperature=TEMPERATURE,
            stream=True,
        )
        for chunk in stream:
            if not chunk.choices:
                continue
            delta = chunk.choices[0].delta.content
            if delta:
                yield delta

    # ---------- 3. 结构化输出（带一次纠错重试）----------
    def structured(
        self,
        messages: list[dict[str, Any]],
        schema: type[T],
        retries: int = 1,
    ) -> T:
        fields = ", ".join(schema.model_fields)
        sys = (
            "只输出一个 JSON 对象，不要 markdown 代码块，不要解释。"
            f"必须包含这些字段：{fields}。字段约束见 schema。"
        )
        msgs = [{"role": "system", "content": sys}, *messages]
        last_err = ""
        for attempt in range(retries + 1):
            if attempt > 0:
                # 关键：把校验错误喂回去，模型能自己修
                msgs.append({
                    "role": "user",
                    "content": f"上一次输出不合法：{last_err}。请只输出修正后的 JSON。",
                })
            resp = self.client.chat.completions.create(
                model=self.provider.model,
                messages=msgs,  # type: ignore[arg-type]
                temperature=0,
                response_format={"type": "json_object"},
            )
            raw = resp.choices[0].message.content or "{}"
            try:
                return schema.model_validate_json(raw)
            except (ValidationError, json.JSONDecodeError) as exc:
                last_err = str(exc)[:400]
        raise RuntimeError(f"结构化输出 {retries + 1} 次均失败: {last_err}")

    ) -> Any:
        return self.client.chat.completions.create(
            model=self.provider.model,
            messages=messages,  # type: ignore[arg-type]
            tools=tools,
            temperature=kw.get("temperature", TEMPERATURE),

这不是洁癖。缓存按前缀匹配，前缀一变缓存全废，成本和首字延迟一起回来。现在写对，W11 和 P3-12 就不用返工。

  • 稳定前缀：system 与工具定义一字不改。绝不把「当前时间」「用户名」「余额」写进 system。
  • 追加末尾：不重排历史、不给消息重新编号、不对历史做「格式化美化」。
  • 标准格式：用 SDK 的消息结构，别自己拼 <|im_start|> 那类模板标记。

W04手写 agent loop：这一阶段的核心

第 4 周 · 60–150 行，不用框架。做完你会有一种「原来就这么点东西」的释然——以及随之而来的困惑「那为什么框架那么复杂」。后者正是 P3 要回答的。

P1-9 工具注册表 + 三个工具（含一个会失败的） 周六 2h
为什么：工具是 agent 唯一的「行动能力」。工具设计好坏直接决定准确率，而设计的核心是 description 和「工具之间有没有重叠」——这一张卡就练这两件事。
关键知识点

模型看不到你的函数，它只看得到 name / description / 参数 schema。 所以工具「能不能被用对」几乎完全由描述决定：写「查询数据库」它不知道什么时候该用；写「按关键词搜公司文档库，返回标题+摘要+URL，支持中文」它才知道该在什么场合调。这就是为什么工具设计（ACI）比工具实现更值钱。

  • schema 的严格程度决定失败率。 参数写成 string，模型自由发挥；写成 enum，选项只有三种。能用 enum 就不用 string，能用扁平参数就不嵌套对象。
  • 失败也要是一个结果。 工具抛异常会让整个 loop 断掉；返回 ok=false + 可读原因，模型才有机会改参数重试（P1-11 展开）。
  • 常见误解：工具越多越强。 工具列表每一轮都要重发，20 个工具的 schema 就上万 token；数量到上百个时，连最强的模型也开始选错（P3-12 有做法和数字）。所以常驻工具要少，其余改成按需加载。

"""src/agentlab/tools.py — 注册表 + 三个工具"""
from __future__ import annotations

import json
import time
from collections.abc import Callable
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any


@dataclass
class ToolResult:
    ok: bool
    content: str
    # 给上层判断用，不进模型上下文
    meta: dict[str, Any] = field(default_factory=dict)


@dataclass
class Tool:
    name: str
    description: str
    parameters: dict[str, Any]
    fn: Callable[..., ToolResult]

    def spec(self) -> dict[str, Any]:
        return {
            "type": "function",
            "function": {
                "name": self.name,
                "description": self.description,
                "parameters": self.parameters,
            },
        }


class Registry:
    def __init__(self) -> None:
        self._tools: dict[str, Tool] = {}

    def register(self, tool: Tool) -> None:
        if tool.name in self._tools:
            raise ValueError(f"工具重名: {tool.name}")
        self._tools[tool.name] = tool

    def specs(self) -> list[dict[str, Any]]:
        return [t.spec() for t in self._tools.values()]

    def names(self) -> list[str]:
        return list(self._tools)


# ---------- 工具 1：只读、快、不会失败 ----------
NOTES_DIR = Path("data/notes")


def search_notes(query: str, limit: int = 3) -> ToolResult:
    """在项目笔记里做关键词搜索。query 用中文或英文均可。"""
    if not NOTES_DIR.exists():
        return ToolResult(ok=False, content="笔记目录不存在")
    hits: list[str] = []
    for p in sorted(NOTES_DIR.rglob("*.md")):
        text = p.read_text(encoding="utf-8", errors="ignore")
        if query.lower() in text.lower():
            idx = text.lower().index(query.lower())
            snippet = text[max(0, idx - 60): idx + 160].replace("\n", " ")
            hits.append(f"[{p.name}] ...{snippet}...")
        if len(hits) >= limit:
            break
    return ToolResult(
        ok=True,
        content="\n".join(hits) if hits else f"没有找到包含「{query}」的内容",
        meta={"n_hits": len(hits)},
    )


# ---------- 工具 2：有副作用、慢、可能超时 ----------
def fetch_url(url: str, timeout: int = 10) -> ToolResult:
    """抓取网页正文的纯文本前 2000 字。仅支持 http/https。"""
    if not url.startswith(("http://", "https://")):
        return ToolResult(ok=False, content="url 必须以 http:// 或 https:// 开头")
    try:
        import httpx

        t0 = time.perf_counter()
        resp = httpx.get(url, timeout=timeout, follow_redirects=True)
        resp.raise_for_status()
        text = resp.text[:2000]
        return ToolResult(
            ok=True,
            content=f"状态 {resp.status_code}，耗时 {(time.perf_counter()-t0)*1000:.0f}ms\n{text}",
            meta={"status": resp.status_code},
        )
    except Exception as exc:  # noqa: BLE001
        # 这里不 raise —— 错误要变成信息回灌给模型
        return ToolResult(ok=False, content=f"抓取失败：{type(exc).__name__}: {exc}")


# ---------- 工具 3：故意会失败，用来测试错误恢复 ----------
def read_file(path: str) -> ToolResult:
    """读取项目内的文本文件。path 必须是相对路径。"""
    p = Path(path)
    if p.is_absolute():
        return ToolResult(ok=False, content="只接受相对路径，例如 notes/todo.md")
    if ".." in p.parts:
        return ToolResult(ok=False, content="不允许访问上级目录")
    if not p.exists():
        siblings = [str(x) for x in p.parent.glob("*")][:5] if p.parent.exists() else []
        # 关键：报错时给出「候选」，模型下一次就能改对
        return ToolResult(
            ok=False,
            content=f"文件 {path} 不存在。同目录下有：{siblings}",
        )
    return ToolResult(ok=True, content=p.read_text(encoding="utf-8")[:4000])

            "type": "object",
            "properties": {
                "query": {"type": "string", "description": "搜索关键词"},

            },

卡住怎么办
  • 「工具描述写不出来」：用固定句式——「做什么 + 什么时候该用 + 什么时候不该用」。第三段最常被漏掉，但它最能减少误调。
  • 「search_notes 和 read_file 会不会重叠」：会。所以 description 里要划清边界：「搜索返回片段，读文件返回全文；要具体内容用 read_file」。
P1-10 loop 核心：消息序列 + 三个停止条件 周六 2h
为什么：这是整个 24 周最重要的 100 行。它一旦写清楚，P3 的 harness、P4 的容错都只是在它上面加东西。
关键知识点

为什么是 loop，不是 chain。 chain（工作流）的步数由人固定：检索→改写→生成，三步就三步。loop 的步数由模型在运行时决定：它觉得信息不够就再调一次工具。这是 agent 和 workflow 的唯一分界线，P3 判断「该做成 agent 还是 workflow」用的就是这条。

  • 步数由模型决定，就必须有兜底出口。 三个出口各治一种死法：模型不再调工具（正常结束）、超时（防单次任务无限拖）、步数用尽（防工具死循环）。缺任一个，就有一种情况会卡死。
  • 最容易漏的一行：把带 tool_calls 的 assistant 消息原样 append 回序列。省掉它，下一轮模型不知道自己调过什么，会重复调同一个工具，或者因为协议不完整直接报错。
  • 隐形成本：工具返回必须截断。一个网页工具返回 200KB，第二轮上下文就爆了，而且这一轮的 token 全部计费——这是 agent 账单失控最常见的原因。

SYSTEM = """你是一个可以使用工具的助手。

规则：

@dataclass
class Step:
    idx: int
    kind: Literal["tool", "final", "stop"]
    name: str = ""

    result: str = ""
    ok: bool = True
    latency_ms: float = 0.0


@dataclass
class RunResult:
    status: Literal["done", "max_steps", "timeout", "error"]
    answer: str
    steps: list[Step]
    messages: list[dict[str, Any]]
    elapsed_ms: float

    @property
    def tool_calls(self) -> int:
        return sum(1 for s in self.steps if s.kind == "tool")


def run(
    task: str,
    llm: LLM,
    registry: Registry,
    max_steps: int = 8,
    timeout_s: float = 60.0,
    max_result_chars: int = 2000,
) -> RunResult:
    t0 = time.perf_counter()
    messages: list[dict[str, Any]] = [
        {"role": "system", "content": SYSTEM},
        {"role": "user", "content": task},
    ]
    steps: list[Step] = []
    tool_budget = 6

    for i in range(max_steps):
        # 停止条件 1：总时长
        if time.perf_counter() - t0 > timeout_s:
            return RunResult("timeout", "任务超时，已中止", steps, messages,
                             (time.perf_counter() - t0) * 1000)

        resp = llm.with_tools(messages, registry.specs())
        msg = resp.choices[0].message

        # 停止条件 2：模型不再调工具 = 给出最终答案
        if not msg.tool_calls:
            answer = msg.content or ""
            steps.append(Step(idx=i, kind="final", result=answer))
            messages.append({"role": "assistant", "content": answer})
            return RunResult("done", answer, steps, messages,
                             (time.perf_counter() - t0) * 1000)

        # 关键：assistant 消息（含 tool_calls）必须原样回填，
        # 否则下一轮模型不知道自己要过什么，会重复调用或报错
        messages.append(msg.model_dump(exclude_none=True))

        for tc in msg.tool_calls:
            if tool_budget <= 0:
                messages.append({
                    "role": "tool",
                    "tool_call_id": tc.id,
                    "content": "工具调用次数已用完，请直接给出最终答案。",
                })
                continue

            name = tc.function.name
            raw_args = tc.function.arguments or "{}"
            ts = time.perf_counter()

            # 参数可能不是合法 JSON —— 这本身就是一个要回灌的错误
            try:
                args = json.loads(raw_args)
            except json.JSONDecodeError as exc:
                result = ToolResult(ok=False, content=f"参数不是合法 JSON：{exc}")
                args = {}
            else:
                tool = registry._tools.get(name)  # noqa: SLF001
                if tool is None:
                    result = ToolResult(
                        ok=False,
                        content=f"没有名为 {name} 的工具。可用：{registry.names()}",
                    )
                else:
                    try:
                        result = tool.fn(**args)
                    except TypeError as exc:
                        result = ToolResult(ok=False, content=f"参数不匹配：{exc}")
                    except Exception as exc:  # noqa: BLE001
                        result = ToolResult(ok=False, content=f"工具异常：{type(exc).__name__}: {exc}")

            tool_budget -= 1
            latency = (time.perf_counter() - ts) * 1000
            steps.append(Step(idx=i, kind="tool", name=name, args=args,
                              result=result.content, ok=result.ok, latency_ms=latency))

            # 长结果截断：不截断会撑爆上下文，这是最常见的隐形成本
            content = result.content
            if len(content) > max_result_chars:
                content = content[:max_result_chars] + f"\n...[已截断，原长 {len(content)} 字符]"

            messages.append({"role": "tool", "tool_call_id": tc.id, "content": content})

    # 停止条件 3：步数用尽
    return RunResult("max_steps", "达到最大步数仍未完成", steps, messages,
                     (time.perf_counter() - t0) * 1000)

为什么：runner 的要求是无交互、可重复、有输出文件。这样后面才能放进 CI、才能对比两次运行。
关键知识点

runner 的职责不是「跑一遍」，是让同一个测试集在改代码前后可比。 所以三件事是硬要求：测试集版本化（jsonl 进 git）、每条 case 相互独立（一条失败不影响其他）、输出结构化（通过率 + 每条明细）。少了任何一条，你拿到的数字就只能自己看，没法拿来论证「这次改动变好了」。

  • 为什么一定要并发：10 条串行 × 每条 20 秒 = 3 分钟。改一行 prompt 要等 3 分钟，第二周就没人愿意跑了。并发到 5 路是 40 秒，这才是能坚持每天跑的量级。
  • 为什么值得做缓存：LLM 不保证幂等，重跑结果会漂。按 case 内容哈希缓存响应，只在改了 prompt 或换了模型时失效——这样「只改了一行 system」也能只重跑受影响的 case。
  • 边界：runner 只给 pass/fail 和明细，不解释为什么。归因是 P2-14 分层诊断的事，两者别混成一件。

import argparse
import json
import time
from collections import Counter

from agentlab.config import PRIMARY
from agentlab.llm import LLM

@dataclass
class CaseResult:
    id: str
    passed: bool
    reason: str
    status: str
    tool_calls: int
    latency_ms: float
    answer: str
    tags: list[str] = field(default_factory=list)


def judge(case: dict, res) -> tuple[bool, str]:
    expect = case.get("expect", {})
    answer = res.answer or ""
    reasons: list[str] = []

    # 1. 终止状态
    if res.status not in ("done",):
        return False, f"未正常结束: {res.status}"

    # 2. 工具选择（只检查「该调的调了」，不检查顺序）
    used = {s.name for s in res.steps if s.kind == "tool"}
    for t in expect.get("tools", []):
        if t not in used:
            reasons.append(f"缺工具 {t}（实际 {sorted(used)}）")

    if "max_tool_calls" in expect and res.tool_calls > expect["max_tool_calls"]:
        reasons.append(f"工具调用 {res.tool_calls} 次，超过 {expect['max_tool_calls']}")

    # 3. 内容断言（大小写不敏感）
    low = answer.lower()
    for kw in expect.get("must_contain", []):
        if kw.lower() not in low:
            reasons.append(f"缺少关键词「{kw}」")
    for kw in expect.get("must_not_contain", []):
        if kw.lower() in low:
            reasons.append(f"出现了不该有的内容「{kw}」")

    return (not reasons), "; ".join(reasons) or "ok"


def main() -> int:
    ap = argparse.ArgumentParser()
    ap.add_argument("--cases", default="evals/v1.jsonl")
    ap.add_argument("--out", default="eval_results/latest.json")
    args = ap.parse_args()

    cases = [json.loads(line) for line in Path(args.cases).read_text().splitlines() if line.strip()]
    llm = LLM(PRIMARY)
    results: list[CaseResult] = []

    for c in cases:
        t0 = time.perf_counter()
        try:
            res = run(c["input"], llm, build_registry())
            passed, reason = judge(c, res)
            status, answer, calls = res.status, res.answer, res.tool_calls
        except Exception as exc:  # noqa: BLE001
            passed, reason = False, f"异常: {type(exc).__name__}: {exc}"
            status, answer, calls = "error", "", 0

        results.append(CaseResult(
            id=c["id"], passed=passed, reason=reason, status=status,
            tool_calls=calls, latency_ms=(time.perf_counter() - t0) * 1000,
            answer=answer[:300], tags=c.get("tags", []),
        ))
        print(f"{'PASS' if passed else 'FAIL'}  {c['id']}  {reason if not passed else ''}")

    n = len(results)
    npass = sum(r.passed for r in results)
    by_tag: dict[str, list[bool]] = {}
    for r in results:
        for t in r.tags:
            by_tag.setdefault(t, []).append(r.passed)

    summary = {
        "n": n,
        "passed": npass,
        "rate": round(npass / n, 3) if n else 0,
        "avg_latency_ms": round(sum(r.latency_ms for r in results) / n) if n else 0,
        "by_tag": {t: f"{sum(v)}/{len(v)}" for t, v in by_tag.items()},
        "failures": [r.id for r in results if not r.passed],
    }
    print(f"\n通过率 {npass}/{n} = {summary['rate']:.0%}  平均耗时 {summary['avg_latency_ms']}ms")
    print("分组：" + "  ".join(f"{k}={v}" for k, v in summary["by_tag"].items()))
    if summary["failures"]:
        print("失败：" + ", ".join(summary["failures"]))

    out = Path(args.out)
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps(
        {"summary": summary, "results": [asdict(r) for r in results]},
        ensure_ascii=False, indent=2), encoding="utf-8")
    print(f"\n已写入 {out}")
    return 0 if npass == n else 1


if __name__ == "__main__":
    raise SystemExit(main())

"""evals/judge.py"""
from __future__ import annotations

from pydantic import BaseModel, Field

from agentlab.llm import LLM

JUDGE_SYSTEM = """你是一个严格的任务完成度评审。

只判断一件事：助手的回答是否完成了用户的任务。
不要因为风格、长度、措辞而扣分。不要因为回答「不够详细」而扣分。

如果任务本身无法完成（比如要求订机票、访问不存在的网站），
助手如实说明「做不到」即为通过。
"""


class Verdict(BaseModel):
    passed: bool = Field(description="是否完成任务")
    reason: str = Field(max_length=200, description="一句话理由，必须引用回答里的具体内容")


def judge_with_llm(llm: LLM, task: str, answer: str) -> Verdict:
    prompt = f"""用户任务：
{task}

助手回答：
{answer}

请判断是否完成任务。"""
    return llm.structured(
        [{"role": "user", "content": prompt}], Verdict, retries=1
    )
然后做一致性检验（这步不能省）
  1. 把 10 条 case 的回答全部导出，你自己先人工标一遍通过/不通过（不看 judge 结果）。
  2. 跑 judge，对比两者。算一个数：一致率。
  3. 对不一致的 case，判断是「judge 错」还是「你错」。多数情况是你错——人工标的时候容易宽松。
  4. 结论写进 notes/eval-judge.md：一致率多少、judge 在什么情况下不可靠。
judge 的三个已知偏差（面试可讲）

Q1 你的 loop 有几个停止条件？分别防什么？

要点：参数名只告诉模型「填什么」，description 告诉它「什么时候该用这个工具」。模型选错工具的成本远高于参数填错——选错是方向错，填错可以重试自愈。

Q4 你写的裸 loop 相比一个生产 harness，缺了哪 5 样东西？

要点：① 状态持久化与中断恢复；② 上下文压缩/裁剪（长任务会爆）；③ 权限与审批（危险工具不能自动执行）；④ 可观测性（每步 trace、成本、延迟）；⑤ 子 agent / 任务分解。（P3 会补全这五样，现在能说出来就说明你知道方向。）

Q5 什么样的任务不该用 agent，用普通脚本就行？

要点：步骤固定、输入输出确定的。比如「每天把 A 目录的 CSV 合并成一个」——写成 cron + 脚本 100% 可靠且免费；用 agent 引入不确定性还花钱。判据：如果步骤能画成固定的流程图，就不要用 agent。举你工作里的真实例子。

Q6 你的 eval 里为什么必须有「应该失败」的 case？

要点：全部 case 都能通过的测试集会给出虚假安全感；should_fail 测的是边界和诚实度（拒答、不越权、抗注入），这些恰是生产事故高发区。

Q7 LLM judge 有哪三个已知偏差？你怎么缓解？

要点：长度偏好（prompt 禁止以长度扣分）、自我偏好（换一家模型做 judge）、位置偏差（A/B 交换位置跑两次）。

Q8 如果 eval 通过率从 60% 提到 90%，但延迟从 1.2s 涨到 4s，你怎么决策？

要点：先问「这是哪类场景」——同步对话场景 4s 已接近用户容忍上限，异步/批处理场景可接受。再问「通过率提升集中在哪些 case」——如果是边缘 case，可能不值得；如果是核心链路，值得配合流式首字优化来缓解感知延迟。关键是给出决策依据，而不是选一个数字。
